# Module drafter (agent S1): DSpark block drafter for Inkling 2 Mini (target: ink::Engine). Static lib `ink_drafter`.
get_filename_component(INK_ROOT ${CMAKE_CURRENT_LIST_DIR}/../.. ABSOLUTE)
add_library(ink_drafter STATIC drafter.cpp attn.cpp fold.cpp)
target_include_directories(ink_drafter PUBLIC ${INK_ROOT}/include PRIVATE ${CMAKE_CURRENT_LIST_DIR})
target_link_libraries(ink_drafter PUBLIC ink_engine ink_kernels_gemv ink_kernels_gemm ink_kernels_misc Threads::Threads)
# fp32 fold/accumulation order is part of the cache contract: no implicit mul+add contraction
target_compile_options(ink_drafter PRIVATE -ffp-contract=off)

set(INK_DR_TESTS ${INK_ROOT}/tests/drafter)
if(EXISTS ${INK_DR_TESTS}/test_drafter.cpp)
  add_executable(test_drafter ${INK_DR_TESTS}/test_drafter.cpp ${INK_DR_TESTS}/ref_drafter.cpp)
  target_include_directories(test_drafter PRIVATE ${CMAKE_CURRENT_LIST_DIR} ${INK_DR_TESTS})
  target_compile_options(test_drafter PRIVATE -ffp-contract=off)
  target_link_libraries(test_drafter ink_drafter)
  add_test(NAME test_drafter COMMAND test_drafter)
endif()
if(EXISTS ${INK_ROOT}/bench/bench_drafter.cpp)
  add_executable(bench_drafter ${INK_ROOT}/bench/bench_drafter.cpp)
  target_include_directories(bench_drafter PRIVATE ${CMAKE_CURRENT_LIST_DIR} ${INK_DR_TESTS})
  target_link_libraries(bench_drafter ink_drafter)
endif()
if(EXISTS ${INK_DR_TESTS}/eval_acc.cpp)
  add_executable(eval_acc ${INK_DR_TESTS}/eval_acc.cpp ${INK_DR_TESTS}/ref_drafter.cpp)
  target_include_directories(eval_acc PRIVATE ${CMAKE_CURRENT_LIST_DIR} ${INK_DR_TESTS})
  target_link_libraries(eval_acc ink_drafter)
endif()
