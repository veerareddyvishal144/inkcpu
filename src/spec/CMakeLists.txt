# Module spec (agent S2): speculative decoding on top of the engine: sampler, adaptive width policy, verify loop (src/spec), tools and tests (tests/spec, bench/bench_spec.cpp).
get_filename_component(INK_ROOT ${CMAKE_CURRENT_LIST_DIR}/../.. ABSOLUTE)
set(INK_SPEC_SRCS "")
foreach(f sampler.cpp policy.cpp spec.cpp)
  if(EXISTS ${CMAKE_CURRENT_LIST_DIR}/${f})
    list(APPEND INK_SPEC_SRCS ${f})
  endif()
endforeach()
if(INK_SPEC_SRCS)
  add_library(ink_spec STATIC ${INK_SPEC_SRCS})
  target_include_directories(ink_spec PUBLIC ${INK_ROOT}/include PRIVATE ${CMAKE_CURRENT_LIST_DIR})
  target_link_libraries(ink_spec PUBLIC ink_engine)
  target_compile_options(ink_spec PRIVATE -ffp-contract=off)
  if(TARGET ink_drafter)
    target_link_libraries(ink_spec PUBLIC ink_drafter)
    target_compile_definitions(ink_spec PUBLIC INK_HAVE_DRAFTER=1)
  endif()
endif()

# M1 tool: tap rows + greedy next token of every position of a token file
if(EXISTS ${CMAKE_CURRENT_LIST_DIR}/dump_taps.cpp)
  add_executable(dump_taps ${CMAKE_CURRENT_LIST_DIR}/dump_taps.cpp)
  target_link_libraries(dump_taps ink_engine)
  target_compile_options(dump_taps PRIVATE -ffp-contract=off)
endif()

# tests (tests/spec/*.cpp): each file is its own executable linked with ink_spec (when it exists) and ink_engine
foreach(t test_verify test_sampler test_policy test_spec test_spec_sampling)
  if(EXISTS ${INK_ROOT}/tests/spec/${t}.cpp AND TARGET ink_spec)
    add_executable(${t} ${INK_ROOT}/tests/spec/${t}.cpp)
    target_include_directories(${t} PRIVATE ${INK_ROOT}/tests/spec ${CMAKE_CURRENT_LIST_DIR})
    target_compile_definitions(${t} PRIVATE INK_SOURCE_DIR="${INK_ROOT}")
    target_compile_options(${t} PRIVATE -ffp-contract=off)
    target_link_libraries(${t} ink_spec)
  endif()
endforeach()
if(EXISTS ${INK_ROOT}/bench/bench_spec.cpp AND TARGET ink_spec)
  add_executable(bench_spec ${INK_ROOT}/bench/bench_spec.cpp)
  target_include_directories(bench_spec PRIVATE ${INK_ROOT}/tests/spec ${CMAKE_CURRENT_LIST_DIR})
  target_compile_definitions(bench_spec PRIVATE INK_SOURCE_DIR="${INK_ROOT}")
  target_link_libraries(bench_spec ink_spec)
endif()
