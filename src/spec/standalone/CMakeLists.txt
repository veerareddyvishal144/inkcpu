# Private top level for agent S2 (the root configure also pulls in modules of other agents that may be mid-edit):
#   PATH=/usr/bin:/bin:$PATH cmake -S src/spec/standalone -B build-s2 -G Ninja -DCMAKE_BUILD_TYPE=Release && nice -n 19 ninja -C build-s2 -j3
# Same flags and module order as the root CMakeLists.txt; adds the drafter module only when its sources exist.
cmake_minimum_required(VERSION 3.20)
project(inkcpu_s2 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_compile_options(-O3 -march=skylake-avx512 -mfma -mf16c -fno-math-errno -Wall -Wextra -Wno-unused-parameter)
get_filename_component(ROOT ${CMAKE_CURRENT_LIST_DIR}/../../.. ABSOLUTE)
include_directories(${ROOT}/include)
enable_testing()
find_package(Threads REQUIRED)
# kernel libraries defined by hand (src/kernels_*/CMakeLists.txt assume the root as CMAKE_SOURCE_DIR)
add_library(ink_kernels_gemv STATIC ${ROOT}/src/kernels_gemv/threadpool.cpp ${ROOT}/src/kernels_gemv/gemv.cpp ${ROOT}/src/kernels_gemv/gemv_ref.cpp)
target_link_libraries(ink_kernels_gemv PUBLIC Threads::Threads)
target_compile_options(ink_kernels_gemv PRIVATE -ffp-contract=off)
add_library(ink_kernels_misc STATIC ${ROOT}/src/kernels_misc/vecmath.cpp ${ROOT}/src/kernels_misc/norms.cpp ${ROOT}/src/kernels_misc/sconv.cpp
  ${ROOT}/src/kernels_misc/router_mix.cpp ${ROOT}/src/kernels_misc/attention.cpp ${ROOT}/src/kernels_misc/reference.cpp)
target_link_libraries(ink_kernels_misc PUBLIC Threads::Threads)
add_library(ink_kernels_gemm STATIC ${ROOT}/src/kernels_gemm/gemm.cpp ${ROOT}/src/kernels_gemm/gemm_small.cpp)
target_link_libraries(ink_kernels_gemm PUBLIC ink_kernels_gemv)
target_compile_options(ink_kernels_gemm PRIVATE -ffp-contract=off)
add_subdirectory(${ROOT}/src/engine engine)
if(EXISTS ${ROOT}/src/drafter/drafter.cpp AND NOT INK_NO_DRAFTER)
  add_subdirectory(${ROOT}/src/drafter drafter)
endif()
add_subdirectory(${ROOT}/src/spec spec)
add_subdirectory(${ROOT}/src/capi capi)
